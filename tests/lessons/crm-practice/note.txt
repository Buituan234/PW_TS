Nhóm chức năng của testInfo
- Lấy thông tin (read only)
title > tên bài test
project.name: tên project
file _.: Đường dẫn file
retry: số lần chạy lại
status: Trạng thái của file test thường dùng ở afterEach
duration: Thời gian chạn, cũng thường dùng ở afterEach

Nhóm chức năng: Điều khiển luồng

- Phạm vi global: tồn tại test - phạm vi toàn bộ file test, chưa tồn tại testInfo
VD: Nếu dùng test.skip thì sẽ skip tất cả các test case bên dưới dòng test.skip
- Phạm vi group: tồn tại test - phạm vi toàn bộ phần test.description, chưa tồn tại testInfo
VD: Nếu dùng test.skip thì sẽ skip toàn bộ số test case bên trong test.description và ở phía dưới của test.skip
- Phạm vi function(trong test case): Tồn tại cả test và testInfo nhưng khuyến khích dùng test Info trong function hơn

import type {
  FullConfig, FullResult, Reporter, Suite, TestCase, TestResult
} from '@playwright/test/reporter';

class MyReporter implements Reporter {
    ///Playwright test bawtsa dadafu chạy  khởi tạo biến đếm (pass =0, fail =0) (before all)
  onBegin(config: FullConfig, suite: Suite) {
    console.log(`Starting the run with ${suite.allTests().length} tests`);
  }

  //Bắt đầu 1 file test  (before each)
  onTestBegin(test: TestCase, result: TestResult) {
    console.log(`Starting test ${test.title}`);
  }


    //afterEach (chạy sau khi file test kết thúc) (nếu test fail => gửi noti qua Msteam......)
  onTestEnd(test: TestCase, result: TestResult) {
    console.log(`Finished test ${test.title}: ${result.status}`);
  }

  // afterALl (chạy hết bài test)=> tính tổng bn test case pass fai. gửi báo ccáo. slack/email
  onEnd(result: FullResult) {
    console.log(`Finished the run: ${result.status}`);
  }
}

export default MyReporter;
